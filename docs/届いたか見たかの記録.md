# 届いたか・見たかを、PIVOT2 から確かめる

作成：2026年9月23日／対象：`ext.gs` と `コード.gs`

---

## 1. なぜ必要か

メールでお送りしていたころは、**届かなければ返ってきた**ので
「送れなかった」がすぐ分かりました。マイページに入れる形では、
**返ってくるものがありません。**

そのかわり、マイページは**もっと強いことが分かります。**
「本当にご覧になったか」です。ただし、**記録しないと何も分かりません。**

| 段 | 何を確かめるか | いま |
|---|---|---|
| 1 | 送った1件ずつ「入ったか」 | **済**（PIVOT2 の表） |
| **2** | **ログインしたか・明細を開いたか** | **これから作ります** |
| 3 | 見ていない方への追いかけ | まだ |

---

## 2. 考えかた

★ **オーナー表に列を足しません。** `コード.gs` の `HEAD` を変えると、
　 読み書きの幅が変わってほかのところに響きます。別の表にします。

| 何を | どこから取るか | 足すもの |
|---|---|---|
| **最終ログイン** | `ログイン試行` シート。**すでに記録があります** | **なし** |
| **明細を開いた日時** | `閲覧` シート（新しく作ります） | `apiPdf` に1行 |

```
ログイン試行（すでにあるもの）
  日時 ／ メール ／ 結果
  2026/9/23 4:07 ／ a@x.jp ／ ログイン     ← これを使います

閲覧（自動でできます）
  メール ／ ファイルID ／ 日時
```

## 3. 貼るもの ①：`ext.gs` の末尾に足す

```javascript
/* ══════════════════════════════════════════════
 *  届いたか・見たかの記録（当社用）
 *
 *  ★なぜ必要か
 *    メールなら、届かなければ返ってきました。マイページでは
 *    返ってくるものがありません。そのかわり「本当にご覧になったか」が
 *    分かります。ただし、記録しないと何も分かりません。
 *
 *  ★オーナー表には列を足しません。
 *    コード.gs の HEAD を変えると読み書きの幅が変わり、
 *    ほかのところに響くためです。別の表にします。
 *
 *  ★最終ログインは、ログイン試行の表にすでに記録があります。
 *    新しい列も、新しい表も要りません。
 * ══════════════════════════════════════════════ */

var SH_SEEN = '閲覧';
var SH_TRY_ = 'ログイン試行';     /* コード.gs が作ります */

function seenSheet_(){
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(SH_SEEN);
  if(!sh){
    sh = ss.insertSheet(SH_SEEN);
    sh.getRange(1, 1, 1, 3).setValues([['メール','ファイルID','日時']]);
    sh.setFrozenRows(1);
  }
  return sh;
}

/* 明細を開いたことを、1行足します。
 * ★失敗しても、明細は出します。記録のために明細が開けないのは本末転倒です。 */
function noteSeen_(email, id){
  try{ seenSheet_().appendRow([norm_(email), String(id || ''), new Date()]); }catch(e){}
}

/* 対象月の数くらべ用（2026年9月 → 202609）。読めなければ 0 */
function ckKey_(ym){
  var v = String(ym == null ? '' : ym);
  try{ if(v.normalize) v = v.normalize('NFKC'); }catch(e){}
  var m = v.match(/(\d{4})\D+(\d{1,2})/);
  if(!m) return 0;
  var y = Number(m[1]), mo = Number(m[2]);
  if(y < 1900 || y > 2200 || mo < 1 || mo > 12) return 0;
  return y * 100 + mo;
}

/* メールごとの「いちばん新しい閲覧日時」と「開いたファイルID」 */
function seenAll_(){
  var out = { last:{}, ids:{} };
  var sh = seenSheet_(), last = sh.getLastRow();
  if(last < 2) return out;
  var v = sh.getRange(2, 1, last - 1, 3).getValues();
  for(var i = 0; i < v.length; i++){
    var m = norm_(v[i][0]);
    var t = extMs_(v[i][2]);
    if(!m) continue;
    if(v[i][1]) out.ids[m + '|' + String(v[i][1])] = 1;
    if(t && (!out.last[m] || t > out.last[m])) out.last[m] = t;
  }
  return out;
}

/* メールごとの「いちばん新しいログイン日時」
 * ★結果が「ログイン」の行だけを見ます。まちがえた行は数えません。 */
function loginAll_(){
  var out = {};
  var sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SH_TRY_);
  if(!sh) return out;
  var last = sh.getLastRow();
  if(last < 2) return out;
  var v = sh.getRange(2, 1, last - 1, 3).getValues();
  for(var i = 0; i < v.length; i++){
    if(String(v[i][2]) !== 'ログイン') continue;
    var m = norm_(v[i][1]);
    var t = extMs_(v[i][0]);
    if(!m || !t) continue;
    if(!out[m] || t > out[m]) out[m] = t;
  }
  return out;
}

/* オーナー様ごとに、届いたか・見たかをまとめます */
function stCheck(req){
  var want = String((req && req.month) || '').trim();   /* '' なら最新月 */

  var osh = ownSheet_();                                 /* 支社の列も用意されます */
  if(!osh || osh.getLastRow() < 2) return { ok:true, list:[] };
  var ov = osh.getRange(2, 1, osh.getLastRow() - 1, COL_OWN_AREA).getValues();

  /* 明細を、メールごとにまとめます */
  var papers = {};
  var psh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('明細');
  if(psh && psh.getLastRow() >= 2){
    var pv = psh.getRange(2, 1, psh.getLastRow() - 1, 3).getValues();
    for(var i = 0; i < pv.length; i++){
      var m = norm_(pv[i][0]);
      if(!m) continue;
      if(!papers[m]) papers[m] = [];
      papers[m].push({ ym:String(pv[i][1] || ''), id:String(pv[i][2] || ''),
                       key:ckKey_(pv[i][1]) });
    }
  }

  var sn = seenAll_(), lg = loginAll_();
  var day = function(t){
    return t ? Utilities.formatDate(new Date(t), 'Asia/Tokyo', 'yyyy/M/d') : '';
  };

  return {
    ok : true,
    list : ov.map(function(r){
      var mail = norm_(r[0]);
      var mine = papers[mail] || [];

      var one = null;
      if(want){
        for(var i = 0; i < mine.length; i++){
          if(mine[i].ym === want){ one = mine[i]; break; }
        }
      }else{
        /* 対象月を指定していなければ、いちばん新しい月を見ます。
         * ★並び順ではなく、月の数で選びます。表は送った順に増えるので、
         *   いちばん下の行が最新月とはかぎりません。 */
        for(var j = 0; j < mine.length; j++){
          if(!one || mine[j].key > one.key) one = mine[j];
        }
      }

      return {
        mail   : mail,
        name   : String(r[1] || ''),
        area   : String(r[COL_OWN_AREA - 1] || ''),
        months : mine.length,
        ym     : one ? one.ym : '',
        pdf    : !!(one && one.id),
        opened : !!(one && one.id && sn.ids[mail + '|' + one.id]),
        login  : day(lg[mail]),
        seen   : day(sn.last[mail])
      };
    })
  };
}

/* 閲覧の記録を間引きます（同じ明細を何度も開くと行が増えるため）。
 * ★手で実行します。ふだんは要りません。 */
function 閲覧の記録を間引く(){
  var sh = seenSheet_(), last = sh.getLastRow();
  if(last < 2){ Logger.log('記録はまだありません。'); return; }
  var v = sh.getRange(2, 1, last - 1, 3).getValues();
  var keep = {};
  for(var i = 0; i < v.length; i++){
    var k = norm_(v[i][0]) + '|' + String(v[i][1] || '');
    var t = extMs_(v[i][2]);
    if(!keep[k] || t > keep[k].t) keep[k] = { t:t, row:i };
  }
  var live = {};
  Object.keys(keep).forEach(function(k){ live[keep[k].row] = 1; });
  var del = 0;
  for(var j = v.length - 1; j >= 0; j--){
    if(!live[j]){ sh.deleteRow(j + 2); del++; }
  }
  Logger.log('間引きました： ' + del + ' 行。のこり ' + (v.length - del) + ' 行。');
}
```

## 7. 確かめかた

1. マイページにログインし、**送金明細の［明細（PDF）を開く］を押す**
2. 台帳に **`閲覧`** という表ができ、1行入っていることを見る
3. PIVOT2 →［オーナーメール］→［オーナー様とのやりとり］の上に
   **［届いたか・見たかを確かめる］**が出ます。押す
4. こう出ます

```
オーナー様   対象月      明細PDF   ログイン   明細を開いた
山田 太郎   2026年9月   あり      9/23      9/23        ● ご覧になっています
鈴木 花子   2026年9月   あり      9/20      —           ▲ 入りましたが未読
佐藤 一郎   —           なし      —         —           ✗ 一度も入っていません
```

## 8. この記録で分かること・分からないこと

| | |
|---|---|
| **分かる** | ログインしたか／その月の明細PDFを開いたか／いつか |
| **分からない** | 中身を読んだか。画面を開いただけかもしれません |
| **分からない** | ご家族や税理士先生がご覧になった場合の区別 |

★ **「メールが返ってこない」より確かな情報です。** メールでは
　 迷惑メール箱に入っても、開かなくても、何も返ってきませんでした。
